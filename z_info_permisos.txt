-- CREATE OR REPLACE VIEW z_info_permisos as
SELECT
	pp.`id_permiso` AS `id_permiso`,
	`e`.`id_empleado` AS `id_empleado`,
	`d`.`lider` AS `lider`,
	`d`.`nombre` AS `departamento`,
	CONCAT(
		`e`.`nombre`, ' ', `e`.`apellido`
	) AS `nombre_empleado`,
	`tp`.`permiso` AS `permiso`,
	pp.`descripcion` AS `descripcion`,
	pp.`fecha_solicitud` AS `fecha_solicitud`,
	pp.`fecha_inicio` AS `fecha_inicio`,
	pp.`fecha_fin` AS `fecha_fin`,
	(
		TIMESTAMPDIFF(
			MINUTE, pp.`fecha_inicio`, pp.`fecha_fin`
		) / 60
	) AS `diferencia`,
	FLOOR(
		(
			(
				SELECT
					`diferencia`
			) / 24
		)
	) AS `dias`,
	(
		TIME_FORMAT(
			TIMEDIFF(
				pp.`fecha_fin`, pp.`fecha_inicio`
			),
			'%H'
		) - (
			24 * (
				SELECT
					`dias`
			)
		)
	) AS `horas`,
	(
		TIME_FORMAT(
			TIMEDIFF(
				pp.`fecha_fin`, pp.`fecha_inicio`
			),
			'%i'
		) * 1
	) AS `minutos`,
	(
		CASE WHEN (pp.`pendiente` = 1) THEN 'Pendiente' WHEN (pp.`aprobado` = 1) THEN 'Aprobado' WHEN (pp.`declinado` = 1) THEN 'Declinado' END
	) AS `estado`,
	(
        SELECT ROUND((TIMESTAMPDIFF(MINUTE,p.fecha_inicio,p.fecha_fin)/60/24) ,1)
        FROM permisos p
        -- JOIN lista_numeros n ON DATE_ADD(p.fecha_inicio, INTERVAL n.n DAY) <= p.fecha_fin
        WHERE p.id_permiso = pp.id_permiso 
    ) AS dias_tomados,  -- Días tomados sin condiciones

    (
        SELECT COUNT(DISTINCT DATE(DATE_ADD(p.fecha_inicio, INTERVAL n.n DAY)))
        FROM permisos p
        JOIN lista_numeros n ON DATE_ADD(p.fecha_inicio, INTERVAL n.n DAY) <= p.fecha_fin
        WHERE p.id_permiso = pp.id_permiso 
          AND DATE(DATE_ADD(p.fecha_inicio, INTERVAL n.n DAY)) NOT IN (SELECT f.fecha FROM festivos f)
    ) AS dias_tomados_sin_festivos,  -- Días tomados sin festivos

    (
        SELECT COUNT(DISTINCT DATE(DATE_ADD(p.fecha_inicio, INTERVAL n.n DAY)))
        FROM permisos p
        JOIN lista_numeros n ON DATE_ADD(p.fecha_inicio, INTERVAL n.n DAY) <= p.fecha_fin
        WHERE p.id_permiso = pp.id_permiso 
          AND (0 = FIND_IN_SET(DAYOFWEEK(DATE(DATE_ADD(p.fecha_inicio, INTERVAL n.n DAY))), 
              (SELECT IFNULL(cc.dias_no_laborales, '') 
               FROM contratos cc 
               WHERE cc.id_empleado = p.id_empleado AND cc.activo = 1)))
    ) AS dias_tomados_sin_dias_no_laborales, 

    (
        SELECT COUNT(DISTINCT DATE(DATE_ADD(p.fecha_inicio, INTERVAL n.n DAY)))
        FROM permisos p
        JOIN lista_numeros n ON DATE_ADD(p.fecha_inicio, INTERVAL n.n DAY) <= p.fecha_fin
        WHERE p.id_permiso = pp.id_permiso 
          AND DATE(DATE_ADD(p.fecha_inicio, INTERVAL n.n DAY)) NOT IN (SELECT f.fecha FROM festivos f)
          AND (0 = FIND_IN_SET(DAYOFWEEK(DATE(DATE_ADD(p.fecha_inicio, INTERVAL n.n DAY))), 
              (SELECT IFNULL(cc.dias_no_laborales, '') 
               FROM contratos cc 
               WHERE cc.id_empleado = p.id_empleado AND cc.activo = 1)))
    ) AS dias_tomados_sin_festivos_y_sin_dias_no_laborales  -- Días sin festivos y sin días no laborables
FROM
	(
		(
			(
				`permisos` pp
				JOIN `empleados` `e` ON(
					(
						`e`.`id_empleado` = pp.`id_empleado`
					)
				)
			)
			JOIN `departamentos` `d` ON(
				(
					`d`.`id_departamento` = `e`.`id_departamento`
				)
			)
		)
		JOIN `tipo_permiso` `tp` ON(
			(
				`tp`.`id_tipo_permiso` = pp.`id_tipo_permiso`
			)
		)
	)
ORDER BY
	pp.`id_permiso` DESC
